# Payments (Stage 5) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Record manual payments against an order (amount + method), derive `orders.payment_status` and per-order dues from the sum of payments, and add a UPI collect flow (in-app QR + WhatsApp share link).

**Architecture:** New `lib/features/payments/` (data/controller/widgets) mirroring the orders feature. A `record_payment` RPC (migration 0012, `security invoker`) inserts a payment and atomically recomputes `orders.payment_status`. The `Order` model gains a joined `payments` list with `paidTotal`/`dues` getters. A `PaymentsController` records then reloads orders. The Order detail grows a payment card with "Record payment" (a form sheet) and "Collect via UPI" (a `qr_flutter` QR + share sheet).

**Tech Stack:** Flutter/Dart, Riverpod `StateNotifier` (auth-scoped like `ordersServiceProvider`), Supabase (`payments` table + `record_payment` RPC), `qr_flutter` (new dep), `url_launcher` (WhatsApp), existing design system (`AppCard`/`AppPrimaryButton`/`StatusPill`/`Money.inr`/tokens).

### Task 1: Migration 0012 — record_payment RPC

**Files:**
- Create: `supabase/migrations/0012_record_payment.sql`

Inserts a payment for the caller's order and recomputes `orders.payment_status` from the sum of payments, atomically. Ownership-checked, invoker-scoped, amount/method validated.

- [ ] **Step 1: Write the migration**

```sql
-- Record a payment against an order and recompute the order's payment_status
-- from the sum of its payments. Invoker-scoped: runs under the caller's RLS,
-- validates inputs, and only ever touches auth.uid()'s own rows.

create or replace function public.record_payment(
  p_order_id uuid,
  p_amount numeric,
  p_method text
) returns void
language plpgsql
security invoker
set search_path = public
as $$
begin
  if p_amount is null or p_amount <= 0 then
    raise exception 'invalid_amount';
  end if;

  if p_method is null or p_method not in ('upi', 'cash', 'other') then
    raise exception 'invalid_method';
  end if;

  if not exists (
    select 1 from orders where id = p_order_id and user_id = auth.uid()
  ) then
    raise exception 'order_not_found';
  end if;

  insert into payments (user_id, order_id, amount, method)
  values (auth.uid(), p_order_id, p_amount, p_method);

  update orders o set payment_status = case
      when coalesce(paid.total, 0) <= 0 then 'unpaid'
      when coalesce(paid.total, 0) >= o.grand_total then 'paid'
      else 'partial'
    end
  from (
    select coalesce(sum(amount), 0) as total
    from payments where order_id = p_order_id
  ) paid
  where o.id = p_order_id and o.user_id = auth.uid();
end;
$$;

revoke all on function public.record_payment(uuid, numeric, text) from public, anon;
grant execute on function public.record_payment(uuid, numeric, text) to authenticated;
```

```dart
/// Builds a UPI intent URI (`upi://pay?...`). The VPA and amount are left raw
/// (the `@` in a VPA is a valid query char; the amount is numeric); the payee
/// name and transaction note are URL-encoded.
String buildUpiUri({
  required String vpa,
  String? name,
  required double amount,
  String? note,
}) {
  final parts = <String>[
    'pa=$vpa',
    'am=${amount.toStringAsFixed(2)}',
    'cu=INR',
  ];
  if (name != null && name.isNotEmpty) {
    parts.add('pn=${Uri.encodeComponent(name)}');
  }
  if (note != null && note.isNotEmpty) {
    parts.add('tn=${Uri.encodeComponent(note)}');
  }
  return 'upi://pay?${parts.join('&')}';
}
```

```dart
import 'package:orderly_app/core/providers/auth_providers.dart';
import 'package:orderly_app/features/orders/controller/orders_provider.dart';

import '../data/payments_service.dart';

final paymentsServiceProvider = Provider<PaymentsService>((ref) {
  ref.watch(authUserIdProvider.select((v) => v.valueOrNull));
  return PaymentsService();
});

final paymentsControllerProvider =
    StateNotifierProvider<PaymentsController, AsyncValue<void>>((ref) {
  return PaymentsController(ref.watch(paymentsServiceProvider), ref);
});

class PaymentsController extends StateNotifier<AsyncValue<void>> {
  PaymentsController(this._service, this._ref)
      : super(const AsyncValue.data(null));

  final PaymentsService _service;
  final Ref _ref;

  /// Records a payment and reloads orders. Returns true on success.
  Future<bool> recordPayment(
    String orderId,
    double amount,
    String method,
  ) async {
    state = const AsyncValue.loading();
    final result = await AsyncValue.guard(
        () => _service.recordPayment(orderId, amount, method));
    state = result.hasError ? result : const AsyncValue.data(null);
    if (result.hasError) return false;
    await _ref.read(ordersControllerProvider.notifier).load();
    return true;
  }
}
```
